require(lmer)
VarComp <- function(.) {
  v <- VarCorr(.)
  s <- sigma(.)^2
  nms <- names(v)
  for (i in nms) {
    if (length(v[[i]])==1) {
      v[[i]] <- c(v[[i]])
    } else {
      # browser()
      c <- attr(v[[i]], "correlation")
      n <- names(diag(v[[i]]))
      nl <- length(n)
      v[[i]] <- c(diag(v[[i]]), attr(v[[i]], "correlation")[lower.tri(diag(nl))])
      ncor <- combn(nl,2)
      names(v[[i]]) <- c(n, apply(ncor,2,function(x)
        paste0("C",paste(n[ x[1] ],n[ x[2] ],sep=":"))) )
    }
  }
  return(c(unlist(v),Residual=s))
}
VarPer <- function(.) {
  v <- VarComp(.)
  v<- v[grep("\\.C",names(v),invert=T,value=T)]
  round(100*v/sum(v),1)
}
