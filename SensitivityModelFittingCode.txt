```{r setup, include=FALSE}
hist.rug <- function(TheVar, IDnames = NULL, ylim=NULL, breaks="Sturges", ...,
                     main=paste("Var=",deparse(substitute(TheVar))), describe=T) {
  if (describe) try(print(psych::describe(TheVar)))
  par(cex.lab = 1, cex.axis = 1.2, lwd = 2)
  suppressWarnings(HistData <- hist(TheVar, plot=F,breaks=breaks, ... ))
  if (is.null(ylim))
    HistData = hist(TheVar, ylim=c(0, max(HistData$counts)*1.2),breaks=breaks, main=main,... )
  else
    HistData = hist(TheVar,breaks=breaks, main=main, ... )
  rug(TheVar, col = "red")
  if (any(class(TheVar)=="Date")) {
    DensityData = density(na.omit(TheVar))
  } else DensityData = density(as.numeric(na.omit(TheVar)))
  lines(DensityData$x, DensityData$y * (max(HistData$counts)/max(DensityData$y)), col = "blue")
  mtext(sprintf("M: %.2f\nSD: %.2f",mean(TheVar,na.rm=TRUE),sd(TheVar,na.rm=TRUE)),side=3, line=-2,adj = 1)

  if (!is.null(IDnames)) identify(TheVar, rep(0, length(TheVar)), labels = IDnames)
}
```

```{r CreateTrainingTest}
set.seed(1312)

CountrySizes <- BTGINTT4 %>% 
  count(CNTRY) %>% 
  mutate(Nover=n-4000) %>% 
  mutate( # select all over 4000 or 10%, whichever is bigger to be test data. this limits training data size for large countries, which was suggested by a reviewer
    Ntrain=ifelse(Nover/n <= 0.1, as.integer(n*0.1) , Nover)) %>% 
  select(-Nover,-n)

BTGINTT4 %>%
  count(CNTRY,IDSCHOOL) %>% 
  merge(CountrySizes) %>% 
  mutate(train=0) %>% 
  group_by(CNTRY) %>% 
  group_modify(~{
    while ( sum(.x$train) < .x$Ntrain[1] ) {
      a <- sample(1:nrow(.x),1)
      .x[a, "train"] <- .x[a,"n"]
    }
    .x
  } ) %>% 
  as.data.frame %>% 
  filter(train>0) %>% 
  with( paste.(CNTRY,IDSCHOOL) ) ->
  TestSchools

Test <- BTGINTT4 %>% filter(paste.(CNTRY,IDSCHOOL) %in% TestSchools )
```

```{r codeconvergeRandom2,eval=T}
YoudenCutoff <- plot(rocit(score=apply(modelRand$yhat.train |> pnorm(),2,mean) ,class=BTGINTT4$Random,negref =0))
macro_predictions <- split( modelRand$yhat.train |> pnorm() %>% apply(c(1), mean) %>% t ,
                            rep(1:modelRand$call$nchain , each=modelRand$call$ndpost/modelRand$call$keepevery) )

mcmc_chains <- lapply(1:length(macro_predictions), function(i) {
  mcmc(macro_predictions[[i]])
}) %>%  as.mcmc.list

gelman.diag(mcmc_chains)
print(coda::effectiveSize(mcmc_chains))
plot(mcmc_chains)


pred_prob_matrix <- modelRand$yhat.train |> pnorm()
# Calculate ESS for each column (teacher) by looking across samples for each case
ess_values <- apply(pred_prob_matrix, 2, effectiveSize)
# Calculate the mean probability for each teacher
mean_probs <- colMeans(pred_prob_matrix)
# Create a data frame combining Teacher ID, Mean Probability, and ESS
teacher_diagnostics <- data.frame(
  IsRandom=modelRand$fit$data@y,
  teacher_id = 1:length(mean_probs),
  mean_prob = mean_probs,
  ess = ess_values  )

cutoff_zone <- teacher_diagnostics[teacher_diagnostics$mean_prob >= YoudenCutoff$optimal[4]-0.05 & 
                                   teacher_diagnostics$mean_prob <= YoudenCutoff$optimal[4]+0.05, ]

teacher_diagnostics |> 
  mutate(mean_prob=cut(mean_prob,seq(0,1,0.01),include.lowest=T) |> factor() |> as.integer() ,
         mean_prob=mean_prob/100,
         IsRandom=factor(IsRandom) ) |> 
  group_by(mean_prob,IsRandom) |> 
  summarize(ess=sum(ess,na.rm=T) , .groups="drop") |> 
  ggplot(aes(x = mean_prob, y = ess,color=IsRandom)) +
  geom_point(alpha = 0.5) +
  geom_vline(xintercept = c(max(YoudenCutoff$optimal[4]-0.05,0), 
                            YoudenCutoff$optimal[4],
                            min(YoudenCutoff$optimal[4]+0.05,1)),
             linetype = "dashed", color = "red") +
  labs(title = "Effective Sample Size vs. Predicted Probability",
       x = "Mean Predicted Probability (Random Country)",
       y = "Effective Sample Size (ESS)") +
  theme_minimal()+ 
  scale_x_continuous(breaks=seq(0,1,0.1),label=scales::percent)+
  scale_y_log10()
```
